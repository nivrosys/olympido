**Stack:** NestJS (TypeScript) · Prisma ORM · PostgreSQL 16 · Redis · BullMQ · Socket.IO  
**Reference:** [`SYSTEM_DESIGN.md`](file:///media/rumon/PLANT/nivrosys/olympido/docs/SYSTEM_DESIGN.md) v2.0.0

## 1. Project Initialization

```bash
# From the olympido root
cd backend
npx -y @nestjs/cli@latest new . --package-manager npm --skip-git --language ts

# Core dependencies
npm i @nestjs/config @nestjs/swagger @prisma/client class-validator class-transformer
npm i @nestjs/passport passport passport-jwt passport-google-oauth20 passport-github2
npm i @nestjs/jwt bcrypt slugify uuid
npm i @nestjs/bullmq bullmq ioredis
npm i @nestjs/websockets @nestjs/platform-socket.io socket.io
npm i @nestjs/throttler
npm i @aws-sdk/client-s3 @aws-sdk/s3-request-presigner
npm i @nestjs/cache-manager cache-manager cache-manager-ioredis-yet
npm i helmet cookie-parser

# Dev dependencies
npm i -D prisma @types/passport-jwt @types/bcrypt @types/uuid
npx prisma init
```

## 3. Prisma Schema (`prisma/schema.prisma`)

```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ============================================================
// Enums
// ============================================================

enum UserRole {
  STUDENT
  TEACHER
  ORGANIZER
  ADMIN
}

enum OrgStatus {
  PENDING_VERIFICATION
  VERIFIED
  REJECTED
  SUSPENDED
}

enum ContestStatus {
  DRAFT
  IN_REVIEW
  CHANGES_REQUESTED
  APPROVED
  SCHEDULED
  LIVE
  EVALUATION
  COMPLETED
  REJECTED
  CANCELLED
}

enum ContestCategory {
  PHYSICS
  MATHEMATICS
  CHEMISTRY
  INFORMATICS
  BIOLOGY
  ASTRONOMY
}

enum DifficultyLevel {
  SCHOOL
  REGIONAL
  NATIONAL
  INTERNATIONAL
}

enum QuestionType {
  MCQ
  SHORT_ANSWER
  THEORETICAL
  CODE
}

enum EvaluationStatus {
  PENDING
  AUTO_GRADED
  MANUAL_REVIEW
  GRADED
}

enum TokenType {
  EMAIL_VERIFICATION
  PASSWORD_RESET
}

// ============================================================
// 1. Users & Authentication
// ============================================================

model User {
  id              String   @id @default(uuid()) @db.Uuid
  email           String   @unique @db.VarChar(255)
  passwordHash    String?  @map("password_hash") @db.VarChar(255)
  role            UserRole @default(STUDENT)
  fullName        String   @map("full_name") @db.VarChar(150)
  username        String   @unique @db.VarChar(50)
  avatarUrl       String?  @map("avatar_url")
  bio             String?
  institutionName String?  @map("institution_name") @db.VarChar(255)
  isEmailVerified Boolean  @default(false) @map("is_email_verified")
  isActive        Boolean  @default(true) @map("is_active")
  createdAt       DateTime @default(now()) @map("created_at") @db.Timestamptz
  updatedAt       DateTime @updatedAt @map("updated_at") @db.Timestamptz

  // Relations
  authProviders          UserAuthProvider[]
  authTokens             AuthToken[]
  teacherOrganizations   TeacherOrganization[]
  verifiedOrganizations  Organization[]        @relation("VerifiedBy")
  createdContests        Contest[]             @relation("Creator")
  reviewedContests       Contest[]             @relation("Reviewer")
  contestRegistrations   ContestRegistration[]
  submissions            Submission[]
  gradedAnswers          SubmissionAnswer[]    @relation("GradedBy")
  contestResults         ContestResult[]
  studentStats           StudentStats?
  posts                  Post[]
  postLikes              PostLike[]
  postComments           PostComment[]
  following              UserFollow[]          @relation("Follower")
  followers              UserFollow[]          @relation("Following")
  notifications          Notification[]
  auditLogs              AuditLog[]
  approvedTeachers       TeacherOrganization[] @relation("ApprovedBy")

  @@map("users")
}

model UserAuthProvider {
  id             String   @id @default(uuid()) @db.Uuid
  userId         String   @map("user_id") @db.Uuid
  provider       String   @db.VarChar(30)
  providerUserId String   @map("provider_user_id") @db.VarChar(255)
  accessToken    String?  @map("access_token")
  refreshToken   String?  @map("refresh_token")
  createdAt      DateTime @default(now()) @map("created_at") @db.Timestamptz

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([provider, providerUserId])
  @@index([userId])
  @@map("user_auth_providers")
}

model AuthToken {
  id        String    @id @default(uuid()) @db.Uuid
  userId    String    @map("user_id") @db.Uuid
  tokenType TokenType @map("token_type")
  tokenHash String    @map("token_hash") @db.VarChar(255)
  expiresAt DateTime  @map("expires_at") @db.Timestamptz
  usedAt    DateTime? @map("used_at") @db.Timestamptz
  createdAt DateTime  @default(now()) @map("created_at") @db.Timestamptz

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId])
  @@index([tokenHash])
  @@map("auth_tokens")
}

// ============================================================
// 2. Organizations & Institutional Verification
// ============================================================

model Organization {
  id                       String    @id @default(uuid()) @db.Uuid
  name                     String    @db.VarChar(255)
  officialDomain           String    @unique @map("official_domain") @db.VarChar(255)
  contactEmail             String    @map("contact_email") @db.VarChar(255)
  websiteUrl               String?   @map("website_url")
  accreditationDocumentUrl String?   @map("accreditation_document_url")
  status                   OrgStatus @default(PENDING_VERIFICATION)
  verifiedById             String?   @map("verified_by") @db.Uuid
  verifiedAt               DateTime? @map("verified_at") @db.Timestamptz
  rejectionReason          String?   @map("rejection_reason")
  createdAt                DateTime  @default(now()) @map("created_at") @db.Timestamptz
  updatedAt                DateTime  @updatedAt @map("updated_at") @db.Timestamptz

  verifiedBy            User?                 @relation("VerifiedBy", fields: [verifiedById], references: [id])
  teacherOrganizations  TeacherOrganization[]
  contests              Contest[]

  @@index([status])
  @@map("organizations")
}

model TeacherOrganization {
  id             String   @id @default(uuid()) @db.Uuid
  userId         String   @map("user_id") @db.Uuid
  organizationId String   @map("organization_id") @db.Uuid
  isApproved     Boolean  @default(false) @map("is_approved")
  approvedById   String?  @map("approved_by") @db.Uuid
  createdAt      DateTime @default(now()) @map("created_at") @db.Timestamptz

  user         User         @relation(fields: [userId], references: [id], onDelete: Cascade)
  organization Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)
  approvedBy   User?        @relation("ApprovedBy", fields: [approvedById], references: [id])

  @@unique([userId, organizationId])
  @@map("teacher_organizations")
}

// ============================================================
// 3. Contests & Admin Approval Workflow
// ============================================================

model Contest {
  id                  String          @id @default(uuid()) @db.Uuid
  creatorId           String          @map("creator_id") @db.Uuid
  organizationId      String?         @map("organization_id") @db.Uuid
  title               String          @db.VarChar(255)
  slug                String          @unique @db.VarChar(255)
  category            ContestCategory
  description         String
  rules               String
  difficulty          DifficultyLevel @default(REGIONAL)
  bannerUrl           String?         @map("banner_url")
  status              ContestStatus   @default(DRAFT)

  // Schedule
  registrationStartAt DateTime  @map("registration_start_at") @db.Timestamptz
  registrationEndAt   DateTime  @map("registration_end_at") @db.Timestamptz
  contestStartAt      DateTime  @map("contest_start_at") @db.Timestamptz
  durationMinutes     Int       @map("duration_minutes")

  // Scoring
  maxScore   Int @default(100) @map("max_score")
  basePoints Int @default(500) @map("base_points")

  // Admin review
  reviewedById     String?   @map("reviewed_by") @db.Uuid
  adminReviewNotes String?   @map("admin_review_notes")
  reviewedAt       DateTime? @map("reviewed_at") @db.Timestamptz

  createdAt DateTime @default(now()) @map("created_at") @db.Timestamptz
  updatedAt DateTime @updatedAt @map("updated_at") @db.Timestamptz

  creator       User          @relation("Creator", fields: [creatorId], references: [id], onDelete: Restrict)
  reviewer      User?         @relation("Reviewer", fields: [reviewedById], references: [id])
  organization  Organization? @relation(fields: [organizationId], references: [id], onDelete: SetNull)
  questions     ContestQuestion[]
  registrations ContestRegistration[]
  submissions   Submission[]
  results       ContestResult[]

  @@index([status])
  @@index([category])
  @@index([creatorId])
  @@index([contestStartAt])
  @@map("contests")
}

// ============================================================
// 4. Contest Questions / Problem Sets
// ============================================================

model ContestQuestion {
  id               String       @id @default(uuid()) @db.Uuid
  contestId        String       @map("contest_id") @db.Uuid
  questionOrder    Int          @map("question_order")
  type             QuestionType @default(MCQ)
  questionText     String       @map("question_text")
  options          Json?        // MCQ: [{"label":"A","text":"...","is_correct":true}]
  correctAnswer    String?      @map("correct_answer")
  explanation      String?
  maxPoints        Decimal      @default(10.00) @map("max_points") @db.Decimal(6, 2)
  timeLimitSeconds Int?         @map("time_limit_seconds")
  attachments      Json?        // [{"url":"...","type":"image"}]
  createdAt        DateTime     @default(now()) @map("created_at") @db.Timestamptz
  updatedAt        DateTime     @updatedAt @map("updated_at") @db.Timestamptz

  contest Contest            @relation(fields: [contestId], references: [id], onDelete: Cascade)
  answers SubmissionAnswer[]

  @@index([contestId])
  @@map("contest_questions")
}

// ============================================================
// 5. Contest Registrations

model ContestRegistration {

  contestId    String   @map("contest_id") @db.Uuid
  userId       String   @map("user_id") @db.Uuid
  registeredAt DateTime @default(now()) @map("registered_at") @db.Timestamptz

  contest Contest @relation(fields: [contestId], references: [id], onDelete: Cascade)
  user    User    @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([contestId, userId])
  @@index([contestId])
  @@index([userId])
  @@map("contest_registrations")
}

// ============================================================
// 6. Submissions & Per-Question Answers
// ============================================================

model Submission {
  id               String           @id @default(uuid()) @db.Uuid
  contestId        String           @map("contest_id") @db.Uuid
  userId           String           @map("user_id") @db.Uuid
  rawScore         Decimal          @default(0.00) @map("raw_score") @db.Decimal(6, 2)
  timeTakenSeconds Int              @map("time_taken_seconds")
  status           EvaluationStatus @default(PENDING)
  submittedAt      DateTime         @default(now()) @map("submitted_at") @db.Timestamptz
  gradedAt         DateTime?        @map("graded_at") @db.Timestamptz

  contest Contest            @relation(fields: [contestId], references: [id], onDelete: Restrict)
  user    User               @relation(fields: [userId], references: [id], onDelete: Restrict)
  answers SubmissionAnswer[]

  @@unique([contestId, userId])
  @@index([contestId])
  @@index([userId])
  @@map("submissions")
}

model SubmissionAnswer {
  id             String    @id @default(uuid()) @db.Uuid
  submissionId   String    @map("submission_id") @db.Uuid
  questionId     String    @map("question_id") @db.Uuid
  answerContent  String?   @map("answer_content")
  answerFileUrl  String?   @map("answer_file_url")
  selectedOption String?   @map("selected_option") @db.VarChar(10)
  score          Decimal   @default(0.00) @db.Decimal(6, 2)
  isCorrect      Boolean?  @map("is_correct")
  gradedById     String?   @map("graded_by") @db.Uuid
  gradedAt       DateTime? @map("graded_at") @db.Timestamptz
  createdAt      DateTime  @default(now()) @map("created_at") @db.Timestamptz

  submission Submission      @relation(fields: [submissionId], references: [id], onDelete: Cascade)
  question   ContestQuestion @relation(fields: [questionId], references: [id], onDelete: Cascade)
  gradedBy   User?           @relation("GradedBy", fields: [gradedById], references: [id])

  @@unique([submissionId, questionId])
  @@index([submissionId])
  @@map("submission_answers")
}
```

## 5. Module Specifications & Key Logic

### 5.1 `PrismaModule` (Global)

```typescript
// prisma.service.ts
@Injectable()
export class PrismaService extends PrismaClient implements OnModuleInit {
  async onModuleInit() { await this.$connect(); }
  async onModuleDestroy() { await this.$disconnect(); }
}
```

- Registered as a **global module** so every service can inject `PrismaService`.

### 5.2 `AuthModule`

| Endpoint | Logic |
|:---------|:------|
| `POST /register/student` | Hash password (bcrypt), create `User` with role `STUDENT`, create `StudentStats` row, send verification email |
| `POST /register/teacher` | Validate org exists & is verified, create `User` with role `TEACHER`, create `TeacherOrganization` (pending), notify admin |
| `POST /register/organizer` | Validate domain (blacklist + MX check), create `Organization` (PENDING), create `User` with role `ORGANIZER`, notify admin |
| `POST /login` | Verify credentials, issue JWT access + refresh tokens |
| `POST /oauth/:provider` | Passport strategy, find-or-create user via `UserAuthProvider`, issue tokens |
| `POST /forgot-password` | Generate hashed token → `AuthToken`, send email with reset link |
| `POST /reset-password` | Validate token hash & expiry, update `passwordHash`, mark token used |
| `POST /verify-email` | Validate token, set `isEmailVerified = true` |
| `POST /refresh` | Validate refresh token, issue new access token |

**Guards applied:** `@Public()` on all auth endpoints (skip JWT guard).

### 5.3 `ContestsModule`

| Endpoint | Logic |
|:---------|:------|
| `GET /contests` | Public list — filter by `status IN (SCHEDULED, LIVE, COMPLETED)`, category, difficulty. Paginated. |
| `GET /contests/:slug` | Include registration count, user's registration status if authenticated |
| `POST /contests` | Roles: `ORGANIZER`, `TEACHER`. Auto-generate slug. Status = `DRAFT` |
| `PUT /contests/:id` | Only allowed in `DRAFT` or `CHANGES_REQUESTED` status. Ownership check. |
| `POST /contests/:id/submit-review` | Validate ≥1 question exists, transition `DRAFT` → `IN_REVIEW` |
| `POST /contests/:id/register` | Validate: status = `SCHEDULED`, registration window open, not already registered |

### 5.4 `QuestionsModule`

- CRUD on `ContestQuestion` — only when contest is in `DRAFT` or `CHANGES_REQUESTED`.
- Reorder: accepts `[{questionId, newOrder}]`, updates `questionOrder` in a transaction.
- During **LIVE** contest: `GET` returns questions **without** `correctAnswer` and `explanation` fields.

### 5.5 `AdminModule`

- All endpoints guarded by `@Roles('ADMIN')`.
- Every action creates an `AuditLog` entry via `AuditLogInterceptor`.
- Contest approval: transition `IN_REVIEW` → `APPROVED` → auto-set `SCHEDULED` if start time is future.
- Org verification: `PENDING_VERIFICATION` → `VERIFIED`, enables contest creation for linked users.

### 5.6 `SubmissionsModule`

| Endpoint | Logic |
|:---------|:------|
| `POST /contests/:id/submit` | Validate: contest is `LIVE`, user is registered, within time window. Create/update `Submission` + `SubmissionAnswer` rows. Enqueue auto-grading job for MCQ/short-answer questions. |
| `PUT /contests/:id/submit/:questionId` | Update a single answer during live contest |
| **Auto-Grading** (BullMQ Job) | Compare `selectedOption` / `answerContent` against `correctAnswer`. Set `isCorrect`, compute `score`. If all questions graded → set submission status = `AUTO_GRADED`. |

### 5.7 `ResultsModule`

| Endpoint | Logic |
|:---------|:------|
| `GET /contests/:id/results` | Paginated finalized rankings |
| `GET /contests/:id/my-results` | Student's own result + per-question breakdown (includes `explanation` after contest) |
| `GET /users/:username/scorecards` | All `ContestResult` rows for user with contest details |
| `GET /verify/scorecard/:hash` | Public verification — returns result data if hash matches |

### 5.8 `LeaderboardsModule` (Redis)

**Redis Keys:**
- `leaderboard:weekly:{YYYY-WW}` — ZSET, score = points earned that week
- `leaderboard:monthly:{YYYY-MM}` — ZSET
- `leaderboard:alltime` — ZSET, score = total_points
- `leaderboard:weekly:{YYYY-WW}:{CATEGORY}` — Subject-filtered ZSET

**Operations:**
- `ZADD` on rating job completion
- `ZREVRANGE` with `WITHSCORES` for leaderboard queries
- `ZREVRANK` for user's current position
- **No periodic reset needed** — weekly keys naturally expire via `EXPIRE` (TTL = 14 days)

### 5.9 `SocialModule`

| Feature | Implementation |
|:--------|:---------------|
| **Posts** | Create post, optionally link to `ContestResult`. Feed = posts from followed users + own, ordered by `createdAt DESC`, cursor-paginated. |
| **Likes** | Toggle: if `PostLike` exists → delete & decrement `likesCount`. Else → create & increment. Uses transaction. |
| **Comments** | Create with optional `parentCommentId` for threading. Increment `commentsCount` on post. |
| **Follows** | Self-follow blocked by DB `CHECK`. Create `UserFollow`, send notification to followed user. |

### 5.10 `NotificationsModule`

- **DB-backed** for persistence + history.
- **WebSocket push** for real-time delivery via `NotificationsGateway`.
- BullMQ `notification` queue handles async creation (prevents blocking main request).
- Types: `CONTEST_APPROVED`, `CONTEST_REJECTED`, `CONTEST_LIVE`, `RATING_CHANGE`, `NEW_FOLLOWER`, `NEW_COMMENT`, `NEW_LIKE`.

### 5.11 `JobsModule` (BullMQ Workers)

| Queue | Processor | Trigger |
|:------|:----------|:--------|
| `auto-grader` | `AutoGraderProcessor` | Submission created |
| `rating-calculation` | `RatingCalculationProcessor` | Contest transitions to `COMPLETED` |
| `contest-scheduler` | `ContestSchedulerProcessor` | Cron: every minute. Checks `SCHEDULED` contests where `contest_start_at <= NOW()` → set `LIVE`. Checks `LIVE` contests past `start + duration` → set `EVALUATION`. |
| `notifications` | `NotificationProcessor` | Any action that triggers a notification |

### 5.12 `ContestGateway` (WebSocket)

## 6. Guards, Interceptors & Middleware

### 6.1 Global JWT Guard

```typescript
// Applied globally in AppModule
// Skipped on routes decorated with @Public()
@Injectable()
export class JwtAuthGuard extends AuthGuard('jwt') {
  canActivate(context: ExecutionContext) {
    const isPublic = this.reflector.getAllAndOverride(IS_PUBLIC_KEY, [...]);
    if (isPublic) return true;
    return super.canActivate(context);
  }
}
```

### 6.2 Roles Guard

```typescript
@Injectable()
export class RolesGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.getAllAndOverride<UserRole[]>(ROLES_KEY, [...]);
    if (!requiredRoles) return true;
    const { user } = context.switchToHttp().getRequest();
    return requiredRoles.includes(user.role);
  }
}
```

### 6.3 Audit Log Interceptor

- Applied to `AdminController`.
- Captures: `actorId`, `action` (derived from route), `targetType`, `targetId`, `oldValue` (pre-query), `newValue` (post-query), `ipAddress`, `userAgent`.
- Writes to `audit_logs` table after successful response.

## 10. Key Design Decisions

| Decision | Rationale |
|:---------|:----------|
| **Prisma over TypeORM** | Type-safe client, superior migration tooling, auto-generated types from schema |
| **BullMQ over Agenda/Celery** | Native Redis-backed, first-class NestJS integration, perfect for rating recalculation |
| **Redis ZSETs for leaderboards** | O(log N) rank queries, natural TTL for weekly/monthly keys, no cron-based resets |
| **Separate `SubmissionAnswer` table** | Per-question tracking enables detailed scorecards and granular analytics |
| **`ON DELETE RESTRICT` on results/submissions** | Historical contest data must never be orphaned; use soft-delete for users |
| **Socket.IO over raw WS** | Room support for contest channels, auto-reconnect, fallback to long-polling |
| **JWT access (15m) + refresh (7d)** | Short-lived access tokens reduce attack surface; refresh stored httpOnly |
| **Audit logs with old/new JSONB** | Full diff capability for admin accountability without complex CDC |
